EVENT TABLE
This is the database table which stores the data about the events in the system and this table created at the time when the admin created the new event in the system.

    id BIGSERIAL PRIMARY KEY,
    -- BASIC
    title VARCHAR(150) NOT NULL,
    subtitle VARCHAR(200),
    description TEXT NOT NULL,
    category VARCHAR(100) NOT NULL,
    event_type VARCHAR(20) NOT NULL, -- free / paid
    entry_fee NUMERIC(10,2),
    tags TEXT[],

    -- ORGANIZER
 We have the coordinators table from where we stores the event id and user id both in that table with coordinating roles.

    -- SCHEDULE
    start_at TIMESTAMP NOT NULL,
    end_at TIMESTAMP NOT NULL,
    event_mode VARCHAR(20) NOT NULL,
    venue VARCHAR(255),
    rooms VARCHAR(255),
    online_link TEXT,
    recurrence VARCHAR(20),

    -- REGISTRATION
    registration_required BOOLEAN DEFAULT TRUE,
    registration_type VARCHAR(30),
    reg_start TIMESTAMP,
    reg_end TIMESTAMP,
    participant_limit INT,
    security_level VARCHAR(20),
    age_restriction INT,

    -- RESOURCES
    accommodation_required BOOLEAN DEFAULT FALSE,
    accommodation_details TEXT,
    accommodation_rooms INT,
    equipment TEXT,
    catering_required BOOLEAN DEFAULT FALSE,
    catering_details TEXT,

    -- TEAM
    teams_enabled BOOLEAN DEFAULT FALSE,
    team_min INT,
    team_max INT,
    team_join_mode VARCHAR(30),
    fee_mode VARCHAR(20),

    -- MEDIA
    poster_url TEXT,
    promo_video TEXT,
    visibility VARCHAR(30),

    -- AUDIENCE
    audience_roles TEXT[],
    audience_course VARCHAR(50),
    audience_department VARCHAR(50),
    audience_years TEXT[],
    external_limit INT,

    -- RESULT
    result_enabled BOOLEAN DEFAULT FALSE,
    result_config JSONB,

    status VARCHAR(20) DEFAULT 'pending',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
    created_by BIGINT REFERENCES users(user_id)






EVENT_FORM_FIELDS TABLE 
This table is used to store the custom fields on the time adding a new event which is used in registration form 
    field_id BIGSERIAL PRIMARY KEY,

    event_id BIGINT REFERENCES events(id) ON DELETE CASCADE,

    label VARCHAR(150) NOT NULL,
    field_type VARCHAR(50) NOT NULL CHECK (
        field_type IN (
            'text',
            'textarea',
            'email',
            'number',
            'tel',
            'url',
            'date',
            'select',
            'checkbox',
            'file'
        )
    ),

    is_required BOOLEAN DEFAULT FALSE,

    options JSONB,  -- for select / checkbox values

    display_order INT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP






















REGISTRATION_CUSTOM_RESPONSES TABLE
This table is used to store the input user data on the time of event registration
    response_id BIGSERIAL PRIMARY KEY,

    registration_id BIGINT REFERENCES event_registrations(registration_id) ON DELETE CASCADE,

    field_id BIGINT REFERENCES event_form_fields(field_id) ON DELETE CASCADE,

    response_value TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    UNIQUE(registration_id, field_id)












EVENT_COORDINATORS TABLE
this is the mapping table which maps coordinators to the events 
    event_id BIGINT REFERENCES events(id) ON DELETE CASCADE,
    user_id BIGINT REFERENCES users(id) ON DELETE CASCADE,
    role VARCHAR(50), -- Primary, Co-Coordinator, Volunteer Head, etc.
    assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY(event_id, user_id)



















USERS TABLE
This table store the user details only necessary details which are mainly common for each type of user 
    user_id BIGSERIAL PRIMARY KEY,

    full_name VARCHAR(120) NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    phone VARCHAR(20),

    password_hash VARCHAR(255) NOT NULL,
    role VARCHAR(30) NOT NULL CHECK (
        role IN (
            'admin',
            'teacher’,         
 	'student',
	'participant'
        )
    ),
    status VARCHAR(20) DEFAULT 'active' CHECK (
        status IN ('pending', 'active', 'inactive', 'blocked')
    )
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP




USER_PROFILES TABLES
This table is the extended details of the users
    user_id BIGINT PRIMARY KEY REFERENCES users(user_id) ON DELETE CASCADE,

    roll_no VARCHAR(50),
    student_id VARCHAR(50),
    year VARCHAR(20),
    branch VARCHAR(100),
    department VARCHAR(120),

    bio TEXT,
    avatar_url TEXT,

    email_alerts BOOLEAN DEFAULT TRUE,
    event_reminders BOOLEAN DEFAULT TRUE,
    team_updates BOOLEAN DEFAULT TRUE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP






EVENT_REGISTRATIONS TABLE
This table stores the user registration for an event 
    registration_id BIGSERIAL PRIMARY KEY,

    event_id BIGINT REFERENCES events(id) ON DELETE CASCADE,
    user_id BIGINT REFERENCES users(user_id) ON DELETE CASCADE,

    team_id BIGINT,  -- NULL for individual registration

    status VARCHAR(20) DEFAULT 'pending' CHECK (
        status IN ('pending', 'approved', 'rejected', 'cancelled', 'waitlisted')
    ),

    submitted_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    approved_at TIMESTAMP,
    cancelled_at TIMESTAMP,

    UNIQUE(event_id, user_id)








EVENT_TEAMS TABLE
This table represents the team as a single entity.
It does NOT represent members.
It does NOT represent registrations.
It represents: “Team Alpha participating in Event X”

    team_id BIGSERIAL PRIMARY KEY,
    event_id BIGINT REFERENCES events(id) ON DELETE CASCADE,
    team_name VARCHAR(150),
    created_by BIGINT REFERENCES users(user_id),
    max_members INT,
    status VARCHAR(20) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP













 EVENT_TEAM_MEMBERS TABLE
This table stores the exact teams id with their roles in the team
    team_member_id BIGSERIAL PRIMARY KEY,
    team_id BIGINT REFERENCES event_teams(team_id) ON DELETE CASCADE,
    user_id BIGINT REFERENCES users(user_id),
    role_in_team VARCHAR(20) DEFAULT 'member',
    joined_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP



















NOTIFICATIONS TABLE
this table stores the notification content, What the message is, Who created it, Which event it belongs to (optional), What type it is
    notification_id BIGSERIAL PRIMARY KEY,

    type VARCHAR(30) CHECK (
        type IN ('approval', 'reminder', 'team', 'announcement', 'system')
    ),

    title VARCHAR(200) NOT NULL,
    message TEXT NOT NULL,

    event_id BIGINT REFERENCES events(id) ON DELETE SET NULL,
    created_by BIGINT REFERENCES users(user_id),

    target_page VARCHAR(100),

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP








USER_NOTIFICATIONS TABLE 
this table stores Who received it, Whether they read it, Whether they archived it

    user_notification_id BIGSERIAL PRIMARY KEY,

    user_id BIGINT REFERENCES users(user_id) ON DELETE CASCADE,
    notification_id BIGINT REFERENCES notifications(notification_id) ON DELETE CASCADE,

    is_read BOOLEAN DEFAULT FALSE,
    is_archived BOOLEAN DEFAULT FALSE,
    is_pinned BOOLEAN DEFAULT FALSE,

    read_at TIMESTAMP,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    UNIQUE(user_id, notification_id)










 EVENT_RESULTS TABLE
Each row of this table stores the result for an event and for multiple positions we used multiple rows 
    result_id BIGSERIAL PRIMARY KEY,

    event_id BIGINT REFERENCES events(id) ON DELETE CASCADE,

    -- Individual OR Team winner
    registration_id BIGINT REFERENCES event_registrations(registration_id),
    team_id BIGINT REFERENCES event_teams(team_id),

    position INT,                     -- 1, 2, 3
    rank_label VARCHAR(50),           -- Winner / Runner-up / 2nd Runner-up

    score NUMERIC(8,2),               -- Final score
    max_score NUMERIC(8,2),           -- Out of total marks

    special_award VARCHAR(150),       -- Best Innovation, Best Design etc.
    remarks TEXT,                     -- Judge comments

    certificate_issued BOOLEAN DEFAULT FALSE,
    certificate_url TEXT,

    declared_by BIGINT REFERENCES users(user_id),
    declared_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
